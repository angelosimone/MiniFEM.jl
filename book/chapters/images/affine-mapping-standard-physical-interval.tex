    \begin{aligned}
      x(\xi)
        &= \frac{a+b}{2}+\frac{b-a}{2}\,\xi, \\[2.4mm]
      \xi(x)
        &= \frac{2x-(a+b)}{b-a}, \\[2.4mm]
      J
        &= \frac{dx}{d\xi}=\frac{b-a}{2}>0, \\[2.4mm]
      dx
        &= J\,d\xi.
    \end{aligned}
